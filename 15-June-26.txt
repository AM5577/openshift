#### install oc client
# 1. Create a dedicated folder for the OpenShift CLI
New-Item -ItemType Directory -Force -Path "C:\OpenShiftCLI"

# 2. Download the latest Windows client zip file
Invoke-WebRequest -Uri "https://downloads-openshift-console.apps.rm3.7wse.p1.openshiftapps.com/amd64/windows/oc.zip" -OutFile "C:\OpenShiftCLI\oc.zip"

# 3. Extract the contents (oc.exe and kubectl.exe)
Expand-Archive -Path "C:\OpenShiftCLI\oc.zip" -DestinationPath "C:\OpenShiftCLI" -Force

# 4. Permanently add the folder to your User PATH
$userPath = [Environment]::GetEnvironmentVariable("Path", "User")
if ($userPath -notmatch "C:\\OpenShiftCLI") {
    [Environment]::SetEnvironmentVariable("Path", "$userPath;C:\OpenShiftCLI", "User")
}
