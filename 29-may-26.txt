####push
podman login
podman push  docker.io/amitfreeze/openshift_microservice:frontend
podman push  docker.io/amitfreeze/openshift_microservice:orderservice
podman push  docker.io/amitfreeze/openshift_microservice:productservice

#login to diff server
#pull
podman pull  docker.io/amitfreeze/openshift_microservice:frontend
podman pull  docker.io/amitfreeze/openshift_microservice:orderservice
podman pull  docker.io/amitfreeze/openshift_microservice:productservice
podman pull  docker.io/amitfreeze/openshift_microservice:userservice

#create contianer network
podman network create microservice_network

#deploy		
podman run -d --name frontend --network microservice_network -p 5000:5000 docker.io/amitfreeze/openshift_microservice:frontend
podman run -d --name orderservice --network microservice_network -p 5003:5003 docker.io/amitfreeze/openshift_microservice:orderservice
podman run -d --name productservice --network microservice_network -p 5002:5002 docker.io/amitfreeze/openshift_microservice:productservice
podman run -d --name userservice --network microservice_network -p 5001:5001 docker.io/amitfreeze/openshift_microservice:userservice
